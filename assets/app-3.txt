const viewport=$('viewport'), world=$('world');

function fitBoard(){
 const r=viewport.getBoundingClientRect();
 const scale=Math.min(r.width/BOARD_W,r.height/BOARD_H)*.965;
 view.scale=scale;view.x=(r.width-BOARD_W*scale)/2;view.y=(r.height-BOARD_H*scale)/2;applyView();
}
function clampView(){
 const r=viewport.getBoundingClientRect(),w=BOARD_W*view.scale,h=BOARD_H*view.scale,margin=70;
 if(w<=r.width){view.x=(r.width-w)/2}else view.x=Math.min(margin,Math.max(r.width-w-margin,view.x));
 if(h<=r.height){view.y=(r.height-h)/2}else view.y=Math.min(margin,Math.max(r.height-h-margin,view.y));
}
function applyView(){clampView();world.style.transform=`translate3d(${view.x}px,${view.y}px,0) scale(${view.scale})`}
function centerOn(id,boost=false){
 const n=M[id],p=pos(n),r=viewport.getBoundingClientRect();
 if(boost)view.scale=Math.max(view.scale,Math.min(r.width/BOARD_W,r.height/BOARD_H)*1.55);
 view.x=r.width/2-p.x*view.scale;view.y=r.height/2-p.y*view.scale;applyView();
}
function zoomAt(factor,cx,cy){
 const old=view.scale, next=Math.max(.22,Math.min(1.25,old*factor));if(next===old)return;
 const wx=(cx-view.x)/old,wy=(cy-view.y)/old;view.scale=next;view.x=cx-wx*next;view.y=cy-wy*next;applyView();
}

 if(e.target.classList.contains('spot')&&e.target.classList.contains('legal'))return;
 viewport.setPointerCapture(e.pointerId);pointers.set(e.pointerId,{x:e.clientX,y:e.clientY});moved=false;
 if(pointers.size===1)dragStart={x:e.clientX,y:e.clientY,vx:view.x,vy:view.y};
 if(pointers.size===2){const a=[...pointers.values()];pinchStart={d:Math.hypot(a[0].x-a[1].x,a[0].y-a[1].y),scale:view.scale,cx:(a[0].x+a[1].x)/2,cy:(a[0].y+a[1].y)/2,x:view.x,y:view.y}}
});
viewport.addEventListener('pointermove',e=>{
 if(!pointers.has(e.pointerId))return;pointers.set(e.pointerId,{x:e.clientX,y:e.clientY});
 if(pointers.size===1&&dragStart){const dx=e.clientX-dragStart.x,dy=e.clientY-dragStart.y;if(Math.abs(dx)+Math.abs(dy)>5)moved=true;view.x=dragStart.vx+dx;view.y=dragStart.vy+dy;applyView()}
 else if(pointers.size===2&&pinchStart){moved=true;const a=[...pointers.values()],d=Math.hypot(a[0].x-a[1].x,a[0].y-a[1].y),cx=(a[0].x+a[1].x)/2,cy=(a[0].y+a[1].y)/2;const next=Math.max(.22,Math.min(1.25,pinchStart.scale*d/pinchStart.d));const wx=(pinchStart.cx-pinchStart.x)/pinchStart.scale,wy=(pinchStart.cy-pinchStart.y)/pinchStart.scale;view.scale=next;view.x=cx-wx*next;view.y=cy-wy*next;applyView()}
});
function pointerEnd(e){pointers.delete(e.pointerId);if(pointers.size<2)pinchStart=null;if(pointers.size===0)dragStart=null}
viewport.addEventListener('pointerup',pointerEnd);viewport.addEventListener('pointercancel',pointerEnd);
